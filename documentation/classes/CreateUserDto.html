import { IsBoolean, IsEmail, IsEnum, IsISO8601, IsNotEmpty, IsOptional, IsString, Matches, MaxLength, MinLength } from "class-validator";
import { UserType } from "../enums/user-type.enums";
import { ApiProperty, ApiPropertyOptional } from "@nestjs/swagger";


/**
 * Create user DTO.
 */
export class CreateUserDto {

    @ApiProperty({
        description: 'The username of the user',
        type: String,
        example: 'johndoe',
    })
    @IsString()
    @MinLength(3)
    @MaxLength(30)
    @IsNotEmpty()
    username: string;

    @ApiProperty({
        description: 'The email of the user',
        type: String,
        example: 'johndoe@email.com',
    })
    @IsEmail()
    @IsNotEmpty()
    email: string;

    @ApiProperty({
        description: 'The password of the user',
        type: String,
        example: 'Kj#9mP$2nL',
    })
    @IsString()
    @IsNotEmpty()
    @MinLength(8)
    @Matches(/^(?=.*[A-Za-z])(?=.*\d)(?=.*[@$!%*#?&])[A-Za-z\d@$!%*#?&]{8,}$/, {
        message:
            'Minimum eight characters, at least one letter, one number and one special character',
    })
    password: string;

    @ApiProperty({
        description: 'The user type of the user',
        enum: UserType,
        example: UserType.USER,
    })
    @IsEnum(UserType)
    @IsNotEmpty()
    userType: UserType;

    @ApiProperty({
        description: 'The date of birth of the user',
        type: String,
        example: '01-01-1990',
    })
    @IsString()
    @IsNotEmpty()
    @MinLength(8)
    @Matches(/^\d{2}-\d{2}-\d{4}$/, {
        message: 'Date must be in DD-MM-YYYY format',
    })
    dateOfBirth: String;

    @ApiPropertyOptional({
        description: 'If the user is active or not',
        type: Boolean,
        example: false,
    })
    @IsBoolean()
    @IsOptional()
    isActive?: boolean;

    @ApiPropertyOptional({
        description: 'If the user is verified or not',
        type: Boolean,
        example: false,
    })
    @IsBoolean()
    @IsOptional()
    isVerified?: boolean;

    @ApiPropertyOptional({
        description: 'If the user is deleted or not',
        type: Boolean,
        example: false,
    })
    @IsBoolean()
    @IsOptional()
    isDeleted?: boolean;

    @ApiPropertyOptional({
        description: 'The created at date of the user',
        type: Date,
        example: '2021-01-01T00:00:00.000Z',
    })
    @IsISO8601()
    @IsOptional()
    createdAt?: Date;

    @ApiPropertyOptional({
        description: 'The updated at date of the user',
        type: Date,
        example: '2021-01-01T00:00:00.000Z',
    })
    @IsISO8601()
    @IsOptional()
    updatedAt?: Date;

    @ApiPropertyOptional({
        description: 'The deleted at date of the user',
        type: Date,
        example: '2021-01-01T00:00:00.000Z',
    })
    @IsISO8601()
    @IsOptional()
    deletedAt?: Date;
